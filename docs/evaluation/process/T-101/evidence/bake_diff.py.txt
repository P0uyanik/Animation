"""Largest per-bone difference between two bakes of the same scene: joint position (skin * rest head), in mm."""
import json, sys
A, B = (json.load(open(p)) for p in sys.argv[1:3])
res = {}
for ca, cb in zip(A["characters"], B["characters"]):
    pkg = json.load(open(ca["package"])); heads = {b["name"]: b["head"] for b in pkg["skeleton"]}
    worst, bones, frames, other = (0, None, None), set(), set(), 0.0
    fb = {f["frame"]: f for f in B["frame_data"]}
    for fa in A["frame_data"]:
        sa, sb = fa["characters"][ca["name"]]["skin"], fb[fa["frame"]]["characters"][cb["name"]]["skin"]
        for i, n in enumerate(ca["bones"]):
            h = heads[n]
            pa = [sum(sa[16*i+c*4+r]*h[c] for c in range(3)) + sa[16*i+12+r] for r in range(3)]
            pb = [sum(sb[16*i+c*4+r]*h[c] for c in range(3)) + sb[16*i+12+r] for r in range(3)]
            d = sum((x-y)**2 for x, y in zip(pa, pb)) ** .5
            if d > 1e-9: bones.add(n); frames.add(fa["frame"])
            if not n.startswith("DYN-"): other = max(other, d)
            if d > worst[0]: worst = (d, n, fa["frame"])
        ma, mb = fa["characters"][ca["name"]].get("morphs", []), fb[fa["frame"]]["characters"][cb["name"]].get("morphs", [])
        if ma != mb: other = max(other, 1e9)
    res[ca["name"]] = {"max_mm": round(worst[0]*1000, 4), "bone": worst[1], "frame": worst[2], "bones_changed": len(bones),
                       "non_DYN_bones_changed": sorted(b for b in bones if not b.startswith("DYN-"))[:5],
                       "frames_changed": len(frames), "max_non_DYN_mm": round(other*1000, 6)}
oa = {k: v for k, v in A.items() if k not in ("frame_data", "characters", "scene_path")}
ob = {k: v for k, v in B.items() if k not in ("frame_data", "characters", "scene_path")}
print(json.dumps(res), "| non-character data identical:", oa == ob)
